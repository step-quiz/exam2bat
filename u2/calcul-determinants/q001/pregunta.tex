Considera les matrius $A=\begin{pmatrix}2&-1\\3&1\end{pmatrix}$ i $B=\begin{pmatrix}1&2\\-1&0\end{pmatrix}$.

\begin{apartats}

\apartat[1,25]{0,75}
Calcula $|A|$, $|B|$, $|A+B|$ i $|A-B|$. És cert que $|A+B|=|A|+|B|$?

\begin{solucio}
$|A|=2\cdot1-(-1)\cdot3=5$ i $|B|=1\cdot0-2\cdot(-1)=2$.\\
$A+B=\begin{pmatrix}3&1\\2&1\end{pmatrix}$, $|A+B|=3-2=1$; $A-B=\begin{pmatrix}1&-3\\4&1\end{pmatrix}$,
$|A-B|=1+12=13$.\\
$|A|+|B|=7\ne1=|A+B|$: el determinant d'una suma no és la suma dels determinants.
\end{solucio}

\begin{tria}{ordre-3}
\itemtria{sarrus}{1}{1,25}
Calcula el determinant de $M=\begin{pmatrix}1&2&-1\\3&0&2\\-2&1&4\end{pmatrix}$.

\begin{solucio}
Per la regla de Sarrus:
\[
|M|=1\cdot0\cdot4+2\cdot2\cdot(-2)+(-1)\cdot3\cdot1-(-1)\cdot0\cdot(-2)-2\cdot3\cdot4-1\cdot2\cdot1
=0-8-3-0-24-2=-37.
\]
\end{solucio}

\itemtria{inventa}{1}{1,25}
Escriu una matriu $2\times2$ sense cap element nul que tingui determinant 1, i una matriu $3\times3$ que
no sigui triangular i que tingui determinant 2. Comprova-ho.

\begin{solucio}
Resposta oberta. Per exemple, $\begin{pmatrix}2&3\\1&2\end{pmatrix}$: $4-3=1$.\\
Per a la $3\times3$, per exemple, $\begin{pmatrix}2&0&0\\1&1&1\\0&1&2\end{pmatrix}$ (no és triangular):
$2\cdot(2-1)-0+0=2$.
\end{solucio}
\end{tria}

\begin{nomesllarg}
\apartat{0,75}
Troba els valors de $x$ que anul·len el determinant $\begin{vmatrix}x-1&2\\3&x\end{vmatrix}$.

\begin{solucio}
$\begin{vmatrix}x-1&2\\3&x\end{vmatrix}=x(x-1)-6=x^2-x-6=(x-3)(x+2)$. S'anul·la per a $x=3$ i per a
$x=-2$.
\end{solucio}
\end{nomesllarg}

\end{apartats}
