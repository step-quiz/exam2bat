Considera els punts $A(0,0,1)$, $B(1,2,0)$ i $C(2,1,1)$.

\begin{apartats}

\apartat[1,25]{0,75}
Troba l'equació general del pla $\pi$ que passa per $A$, $B$ i $C$.

\begin{solucio}
$\overrightarrow{AB}=(1,2,-1)$ i $\overrightarrow{AC}=(2,1,0)$.
\[
\begin{vmatrix}x&y&z-1\\1&2&-1\\2&1&0\end{vmatrix}=x-2y-3(z-1)=0
\ \Longrightarrow\ \pi\colon x-2y-3z+3=0.
\]
Comprovació: $B$ dona $1-4-0+3=0$, i $C$, $2-2-3+3=0$.
\end{solucio}

\begin{tria}{pla-punts}
\itemtria{coplanaris}{1}{1,25}
Si $D$ és el punt $(k,0,0)$, quant ha de valer $k$ perquè $A$, $B$, $C$ i $D$ siguin coplanaris? Són
coplanaris $A$, $B$, $C$ i $E(1,1,1)$?

\begin{solucio}
$D$ ha de ser del pla $\pi$: $k+3=0$, $k=-3$.\\
Per a $E$: $1-2-3+3=-1\ne0$. $E$ no és del pla $\pi$, i els quatre punts no són coplanaris.
\end{solucio}

\itemtria{recta-i-punt}{1}{1,25}
Troba l'equació del pla que conté la recta $r\colon\dfrac{x-1}{1}=\dfrac{y}{2}=\dfrac{z+1}{-1}$ i el punt
$P(0,1,2)$.

\begin{solucio}
El pla passa per $R(1,0,-1)$, un punt de $r$, i té com a vectors directors el de la recta,
$(1,2,-1)$, i $\overrightarrow{RP}=(-1,1,3)$, que no hi és proporcional.
\[
\begin{vmatrix}x-1&y&z+1\\1&2&-1\\-1&1&3\end{vmatrix}=7(x-1)-2y+3(z+1)=0
\ \Longrightarrow\ 7x-2y+3z-4=0.
\]
Comprovació: $P$ dona $0-2+6-4=0$.
\end{solucio}
\end{tria}

\begin{nomesllarg}
\apartat{0,75}
Troba els punts on el pla $\pi$ talla els eixos de coordenades.

\begin{solucio}
Eix $OX$: $x+3=0$, $(-3,0,0)$. Eix $OY$: $-2y+3=0$, $\left(0,\frac32,0\right)$. Eix $OZ$: $-3z+3=0$,
$(0,0,1)$, que és el punt $A$.
\end{solucio}
\end{nomesllarg}

\end{apartats}
