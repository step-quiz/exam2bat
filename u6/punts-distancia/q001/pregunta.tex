Considera la recta $r\colon\left\{\begin{aligned}x&=1+\lambda\\y&=2-\lambda\\z&=2\lambda\end{aligned}\right.$
i el pla $\pi\colon 2x+y-2z-1=0$.

\begin{apartats}

\apartat[1,25]{0,75}
Comprova que $r$ talla el pla $\pi$ i troba el punt de tall.

\begin{solucio}
$\vec u=(1,-1,2)$ i $\vec n=(2,1,-2)$: $\vec u\cdot\vec n=2-1-4=-3\ne0$, i la recta talla el pla.
Substituint: $2(1+\lambda)+(2-\lambda)-4\lambda-1=3-3\lambda=0$, $\lambda=1$: el punt $(2,1,2)$.
\end{solucio}

\begin{tria}{a-distancia-2}
\itemtria{punts-recta}{1}{1,25}
Troba els punts de la recta $r$ que són a distància 2 del pla $\pi$.

\begin{solucio}
Un punt de $r$ és $(1+\lambda,\ 2-\lambda,\ 2\lambda)$, i $|\vec n|=\sqrt{4+1+4}=3$:
\[
\frac{|2(1+\lambda)+(2-\lambda)-4\lambda-1|}{3}=\frac{|3-3\lambda|}{3}=|1-\lambda|=2.
\]
$1-\lambda=2$ dona $\lambda=-1$, i $1-\lambda=-2$, $\lambda=3$. Els punts són $(0,3,-2)$ i $(4,-1,6)$.
\end{solucio}

\itemtria{plans-a-distancia}{1}{1,25}
Troba els plans paral·lels a $\pi$ que són a distància 2 de $\pi$.

\begin{solucio}
Són de la forma $2x+y-2z+D=0$. La distància entre plans paral·lels, amb el punt $(0,1,0)$ de $\pi$:
$\dfrac{|1+D|}{3}=2$, és a dir, $1+D=\pm6$: $D=5$ o $D=-7$. Els plans són $2x+y-2z+5=0$ i
$2x+y-2z-7=0$.
\end{solucio}
\end{tria}

\begin{nomesllarg}
\apartat{0,75}
Troba els punts de l'eix $OX$ que són a distància 2 del pla $\pi$.

\begin{solucio}
Un punt de l'eix és $(x,0,0)$: $\dfrac{|2x-1|}{3}=2$, $2x-1=\pm6$. Els punts són
$\left(\frac72,0,0\right)$ i $\left(-\frac52,0,0\right)$.
\end{solucio}
\end{nomesllarg}

\end{apartats}
